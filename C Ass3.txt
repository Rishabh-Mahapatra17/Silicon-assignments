/*To swap two variables without using a third variable*/
#include<stdio.h>
int main()
{
    int num1, num2;
    printf("Enter any two numbers: ");
    scanf("%d %d", &num1, &num2);
    printf("\nBefore swapping num1=%d and num2=%d", num1, num2);

    num1=num1+num2;
    num2=num1-num2;
    num1=num1-num2;
    printf("\nAfter swapping num1=%d and num2=%d", num1, num2);
}

{

   scanf("%d", &num);

   d1=num%10;
   d2=(num/10)%10;
   d3=num/100;

   sum=(d1+d2+d3);
   printf("\nSum=%d", sum);
}

/*Farenheit to Celsius*/
#include <stdio.h>

int main() {
    float c, f;

    printf("Enter temperature in Fahrenheit: ");
    scanf("%f", &f);

    c = (f-32)*5.0/9.0;//C=F-32*5/9

    printf("%f Fahrenheit = %f Celsius\n", f, c);
}

/*Celsius to Fahrenheit*/
#include <stdio.h>

int main() {
    float f, c;

    printf("Enter temperature in Celsius: ");
    scanf("%f", &c);

    f = (c*9.0/5.0)+32.0;//F=C*9/5+32

    printf("%f Celsius = %f Fahrenheit\n", c, f);
}

#pr11
/*To find the average marks of 5 subjects*/
#include <stdio.h>
int main()
{
    float sub1, sub2, sub3, sub4, sub5, avg;
    printf("Enter the marks for 5 subjects: ");
    scanf("%f %f %f %f %f", &sub1, &sub2, &sub3, &sub4, &sub5);

    avg=(sub1+sub2+sub3+sub4+sub5)/5;
    printf("\nThe average marks of 5 subjects is %f", avg);
}

/*To calculate the gross salary*/
#include <stdio.h>
int main()
{
    float basic, da, hra, gross;
    printf("Enter the basic salary: ");
    scanf("%f", &basic);

    da=0.45*basic;
    hra=0.30*basic;
    gross=basic+da+hra;

    printf("\nThe gross salary is %f.", gross);
}

int main()
{
    int sec,hr,min;
    printf("Enter the time value in seconds: ");
    scanf("%d", &sec);

    hr=sec/3600;
    min=(sec%3600)/60;
    sec=(sec%3600)%60;

    printf("\nThe time is %d hours, %d minutes, %d seconds", hr, min, sec);
}

#pr15
/*To find the largest between three numbers*/
#include <stdio.h>
int main()
{
    int num1, num2, num3, largest;
    printf("Enter the three numbers: ");
    scanf("%d %d %d", &num1, &num2, &num3);

    largest=(num1>num2)?(num1>num3?num1:num3):(num2>num3?num2:num3);
    printf("\nThe largest number is %d.", largest);
}
